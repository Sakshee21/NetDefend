    pkts      bytes target     prot opt in     out     source               destination

[h5 10.0.0.5] $ /usr/sbin/nft list ruleset  (exit 0)
(empty)

[h6 10.0.0.6] $ /usr/sbin/iptables -S  (exit 0)
-P INPUT ACCEPT
-P FORWARD ACCEPT
-P OUTPUT ACCEPT
-A INPUT -s 10.0.0.5/32 -p tcp -m tcp --dport 8000 -j DROP

[h6 10.0.0.6] $ /usr/sbin/iptables -L INPUT -v -n -x  (exit 0)
Chain INPUT (policy ACCEPT 0 packets, 0 bytes)
    pkts      bytes target     prot opt in     out     source               destination         
      20     1200 DROP       tcp  --  *      *       10.0.0.5             0.0.0.0/0            tcp dpt:8000

[h6 10.0.0.6] $ /usr/sbin/nft list ruleset  (exit 0)
table ip filter {
	chain INPUT {
		type filter hook input priority filter; policy accept;
		meta l4proto tcp ip saddr 10.0.0.5 tcp dport 8000 counter packets 20 bytes 1200 drop
	}
}
